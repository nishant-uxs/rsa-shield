\begin{frame}{Problem (epistemic, not new math)}
  \begin{itemize}
    \item Scanners often treat ``short probe failed'' as PASS / SECURE
    \item Bounded failure to factor $\neq$ proof of security
    \item Legacy / embedded RSA still appears in ICS and enterprise PKI
    \item Need a defensive auditor with honest evidence tiers
  \end{itemize}
\end{frame}

\begin{frame}{What we built}
  \begin{itemize}
    \item FastAPI + React auditor (PEM / $(N,e)$; private keys rejected)
    \item Probes: analyzer, Fermat, Wiener, Pollard $p{-}1$, educational ROCA, ECM, batch GCD
    \item Labels: \texttt{FACTOR\_FOUND} / \texttt{SUSPICIOUS} / \texttt{INCONCLUSIVE} / \texttt{NO\_WEAKNESS\ldots} / \texttt{INVALID}
    \item \textbf{Never} \texttt{SECURE}
    \item RSA-Auditor Bench v1.0 + frozen JSON evaluation dumps
  \end{itemize}
\end{frame}

\begin{frame}{ROCA calibration vs badkeys}
  \begin{itemize}
    \item Legacy educational fingerprint over-triggered on modern CT moduli
    \item Calibration set $n{=}500$: FP $375/500$ ($75\%$), agreement $25\%$
    \item Chosen profile: $\texttt{ratio\_threshold}{=}0.99$, $\texttt{min\_gcd}{=}2$
    \item After calibration: FP $1/500$ ($0.2\%$), agreement $99.8\%$
    \item Independent parity $n{=}400$: calibrated agreement $100\%$
    \item Positives remain \texttt{SUSPICIOUS} only (fingerprint $\neq$ factorization)
  \end{itemize}
\end{frame}

\begin{frame}{Adaptive probe scheduler}
  \begin{itemize}
    \item Fixed budgets waste work on $2048$-bit CT-like keys
    \item Planner shrinks/skips Fermat; trims Pollard/ECM for large bit-lengths
    \item Keeps aggressive budgets for small/legacy keys and large-$e$ Wiener cases
    \item Bench: same planted-factor recall ($1.0$), lower mean latency ($1.511\times$)
  \end{itemize}
\end{frame}

\begin{frame}{Public CT tip sample}
  \begin{itemize}
    \item Logs: Argon2026h2, Xenon2026h2, Nimbus2026
    \item $19504$ leaf entries $\to$ $12010$ unique RSA moduli in $1586.86$\,s
    \item Bit lengths: $11654\times 2048$, $336\times 4096$, $20\times 3072$
    \item Chunked batch GCD: $0$ shared-factor hits in $81172.63$\,ms
    \item Tip sample $\neq$ billion-certificate census
  \end{itemize}
\end{frame}

\begin{frame}{Limitations (say these out loud)}
  \begin{itemize}
    \item Not a new general factorization algorithm
    \item CT study is a tip sample, not a full Internet census
    \item \texttt{badkeys} is a fingerprint reference, not Infineon ground truth
    \item Authorized public keys only; inconclusive $\neq$ certification
  \end{itemize}
\end{frame}
